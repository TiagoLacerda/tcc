\begin{algorithm}[!ht]
\begin{footnotesize}
\SetKwInOut{Input}{entrada}
\SetKwInOut{Output}{saída}
\SetAlgoLined
\Input{Grafo conexo $G(V,E)$,\\$start$,\\$end$}
\Begin
{
    $n \gets |V(G)|$\;
    $m \gets |E(G)|$\;
    $candidate \gets$ Grafo vazio de tamanho $n$\;
    $pointers \gets$ Vetor de tamanho $n$\;
    $edges \gets$ Vetor de arestas de $G$\;
    $p = 0$\;
    $pointers[p] = start - 1$\;
    \While{$pointers[0] < end$}
    {
        $pointers[p] \gets pointers[p] + 1$\;
        \uIf{$pointers[p] < m$}
        {
            $candidate.insert(edges[pointers[p]])$\;
            \If{$p == n - 2$}
            {
                \If {$candidate.acyclic()$}
                {
                    // Encontrou uma árvore geradora
                }
                 $candidate.remove\_edge(edges[pointers[p]])$\;
            }
        }
        \Else
        {
            $pointers[p] \gets pointers[p] - 1$\;
            \If{$p > -1$}
            {
                $candidate.remove\_edge(edges[pointers[p]])$\;
            }
        }    
    }
}
\caption{\textsc{$generate\_parallel$}\label{alg:generate_parallel}}
\end{footnotesize}
\end{algorithm}
