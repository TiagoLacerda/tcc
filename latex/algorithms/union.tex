\begin{algorithm}[!ht]
\begin{footnotesize}
\SetKwInOut{Input}{entrada}
\SetAlgoLined
\Input{índice $u$ de um elemento,\\índice $v$ de um elemento}
\Begin
{
    $u \gets find(u)$\;
    $v \gets find(v)$\;
    \If{$u == v$}
    {
        \Return\;
    }
    \uIf{$rank[u] < rank[v]$}
    {
        $parent[u] \gets v$\;
    }
    \uElseIf{$rank[v] < rank[u]$}
    {
        $parent[v] \gets u$\;
    }
    \Else
    {
        $parent[u] \gets v$\;
        $rank[v] \gets rank[v] + 1$\;
    }
}
\caption{\textsc{$union$}\label{alg:union}}
\end{footnotesize}
\end{algorithm}
